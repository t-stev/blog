{{- /* Long-side photo sizing (matches Ink's editor): big landscapes span the column, big portraits take two-thirds of it, both centered. Dimensions come from the page-bundle resource; Ink publishes images with EXIF rotation baked in, so Width/Height are trustworthy. */ -}} {{- $img := .Page.Resources.GetMatch .Destination -}} {{- $class := "" -}} {{- if $img -}} {{- if ge (math.Max $img.Width $img.Height) 700 -}} {{- $class = cond (gt $img.Height $img.Width) "portrait" "landscape" -}} {{- end -}} {{- end -}} {{- /* Ink stores its aspect ratio in the alt slot ("1.00") — drop those */ -}} {{- $alt := .Text -}} {{- if findRE `^\d+(\.\d+)?$` $alt -}}{{- $alt = "" -}}{{- end -}} {{- if $img -}} {{- /* Build-time responsive sizes: browsers pick the smallest sufficient copy. The original is the largest candidate and the src fallback. */ -}} {{- $srcset := slice -}} {{- range $w := slice 480 800 1200 -}} {{- if lt $w $img.Width -}} {{- $r := $img.Resize (printf "%dx q80" $w) -}} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink $w) -}} {{- end -}} {{- end -}} {{- $srcset = $srcset | append (printf "%s %dw" $img.RelPermalink $img.Width) -}} {{- /* Text column is 42rem (~672px); portraits render at 66% of it */ -}} {{- $sizes := cond (eq $class "portrait") "(max-width: 720px) 66vw, 444px" "(max-width: 720px) 100vw, 672px" -}} {{- else -}} {{- end -}}
